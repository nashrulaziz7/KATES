// Setup tombol kelas dengan penanganan event yang bersih document.querySelectorAll('.class-btn').forEach(btn => { const key = btn.id.split('-')[1]; let lpTimer; let isLongPress = false; // Mencegah klik kanan menu browser muncul btn.addEventListener('contextmenu', e => e.preventDefault()); const startAction = (e) => { isLongPress = false; lpTimer = setTimeout(() => { isLongPress = true; if(navigator.vibrate) navigator.vibrate(50); bukaSettings(key); }, 800); // Tahan 0.8 detik untuk edit }; const endAction = (e) => { clearTimeout(lpTimer); if (!isLongPress) { pilihKelas(key); } }; btn.addEventListener('mousedown', startAction); btn.addEventListener('mouseup', endAction); btn.addEventListener('mouseleave', () => clearTimeout(lpTimer)); btn.addEventListener('touchstart', (e) => { startAction(); }, {passive: true}); btn.addEventListener('touchend', endAction); btn.addEventListener('touchmove', () => clearTimeout(lpTimer)); }); // Set default ke JUMLAH saat pertama dimuat setMode('jumlah'); // Inisialisasi awal UI kelas renderTombolKelas();
